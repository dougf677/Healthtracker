create table health_tracker_entries ( user_id uuid primary key references auth.users(id) on delete cascade, entries jsonb not null default '[]'::jsonb, updated_at timestamptz default now() ); alter table health_tracker_entries enable row level security; create policy "Users can read own health tracker entries" on health_tracker_entries for select using (auth.uid() = user_id); create policy "Users can insert own health tracker entries" on health_tracker_entries for insert with check (auth.uid() = user_id); create policy "Users can update own health tracker entries" on health_tracker_entries for update using (auth.uid() = user_id) with check (auth.uid() = user_id);
